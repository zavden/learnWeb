```svg

  <rect x="0" y="0" width="360" height="240" fill="#ffffff" />

  <line x1="40" y1="200" x2="320" y2="200" stroke="#64748b" stroke-width="2" />
  <line x1="40" y1="200" x2="40" y2="30" stroke="#64748b" stroke-width="2" />

  <text id="legend" x="48" y="34" fill="#0f172a" font-size="14" font-family="monospace">random points</text>

```

# JavaScript

```javascript
const svg = document.getElementById('plotter');
const layer = document.getElementById('points-layer');
const legend = document.getElementById('legend');

function drawPoints() {
  layer.innerHTML = '';

  const points = Array.from({ length: 8 }, (_, index) => ({
    x: 60 + Math.random() * 240,
    y: 60 + Math.random() * 120,
    color: index % 2 === 0 ? '#58a6ff' : '#f0883e',
  }));

  points.forEach((point, index) => {
    const circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    circle.setAttribute('cx', point.x.toFixed(1));
    circle.setAttribute('cy', point.y.toFixed(1));
    circle.setAttribute('r', '6');
    circle.setAttribute('fill', point.color);
    layer.appendChild(circle);

    const label = document.createElementNS('http://www.w3.org/2000/svg', 'text');
    label.setAttribute('x', (point.x + 10).toFixed(1));
    label.setAttribute('y', (point.y + 4).toFixed(1));
    label.setAttribute('font-size', '11');
    label.setAttribute('font-family', 'monospace');
    label.setAttribute('fill', '#0f172a');
    label.textContent = `${index + 1}`;
    layer.appendChild(label);
  });

  legend.textContent = 'click chart to randomize';
}

svg.addEventListener('click', drawPoints);
drawPoints();
```
