# CSS

body {
  display: grid;
  place-items: center;
  min-height: 100vh;
  margin: 0;
  background:
    radial-gradient(circle at top, rgba(88, 166, 255, 0.2), transparent 40%),
    linear-gradient(180deg, #020617 0%, #0f172a 100%);
}

#pulse-demo {
  width: min(560px, 90vw);
  cursor: pointer;
}

.halo {
  fill: rgba(88, 166, 255, 0.12);
  stroke: rgba(88, 166, 255, 0.5);

  transform-origin: center;
  transition: transform 220ms ease, opacity 220ms ease;

  fill: #f0e050;
  transition: transform 220ms ease;

}

}
