# CSS

body {
  margin: 0;
  min-height: 100vh;
  display: grid;
  place-items: center;
  background: radial-gradient(circle at top, rgba(88, 166, 255, 0.16), transparent 32%), #020617;
}

#grid-lab {
  width: min(700px, 92vw);
}

.handle {
  fill: #f0e050;
  stroke: #0f172a;
  stroke-width: 2;
  transition: transform 160ms ease, fill 160ms ease;
}

#grid-lab.is-animated .handle:nth-child(odd) {
  transform: translateY(-10px);
  fill: #39d2c0;
}

#grid-lab.is-animated .handle:nth-child(even) {

# JavaScript

```javascript
const lab = document.getElementById('grid-lab');

lab.addEventListener('mouseenter', () => {
  lab.classList.add('is-animated');
});

lab.addEventListener('mouseleave', () => {
  lab.classList.remove('is-animated');
});
```
