# JSX
```jsx
function App() {
  const [active, setActive] = React.useState('Docs');

  return (
    <main className="overview-card">
      <p className="eyebrow">React overview</p>
      <h1>{active}</h1>
      <div className="pill-row">
        {['Docs', 'Hooks', 'Projects'].map((item) => (
          <button key={item} type="button" onClick={() => setActive(item)}>
            {item}
          </button>
        ))}
      </div>
    </main>
  );
}
```
