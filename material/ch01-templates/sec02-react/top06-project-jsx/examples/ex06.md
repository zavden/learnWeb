## @role context

```jsx
import React from 'react';

const LessonContext = React.createContext([]);

const defaultLessons = [
  { id: 'component', title: 'component card', level: 'Core', summary: 'Reusable UI lives in component files.' },
  { id: 'hook', title: 'hook logic', level: 'Shared', summary: 'Hooks keep stateful logic reusable.' },
  { id: 'page', title: 'page shell', level: 'Flow', summary: 'Pages compose multiple smaller files.' },
];

export function LessonsProvider({ children }) {
  return (
    <LessonContext.Provider value={defaultLessons}>
      {children}
    </LessonContext.Provider>
  );
}

export function useLessons() {
  return React.useContext(LessonContext);
}
```

## @file src/hooks/useLessonProgress.js
## @lang javascript
## @role hook

```javascript
import React from 'react';

export function useLessonProgress(total) {
  const [reviewed, setReviewed] = React.useState(0);

  function reviewNext() {
    setReviewed((value) => Math.min(total, value + 1));
  }

  return {
    progressLabel: `${reviewed}/${total} templates reviewed`,
    reviewNext,
    reviewed,
  };
}
```

## @file src/reducers/filterReducer.js
## @lang javascript
## @role reducer

```javascript
export const initialFilterState = {
  level: 'All',
};

export function filterReducer(state, action) {
  if (action.type === 'set-level') {
    return { ...state, level: action.level };
  }

  return state;
}
```

## @file src/utils/formatLesson.js
## @lang javascript
## @role util

```javascript
export function formatLesson(value) {
  return value
    .split(' ')
    .map((part) => part.charAt(0).toUpperCase() + part.slice(1))
    .join(' ');
}
```

## @file src/pages/LessonsPage.jsx
## @lang jsx
## @role page

```jsx
import React from 'react';
import { LessonCard } from '../components/LessonCard.jsx';
import { useLessons } from '../context/LessonContext.jsx';
import { useLessonProgress } from '../hooks/useLessonProgress.js';
import { initialFilterState, filterReducer } from '../reducers/filterReducer.js';
import { formatLesson } from '../utils/formatLesson.js';

export function LessonsPage() {
  const lessons = useLessons();
  const [filter, dispatch] = React.useReducer(filterReducer, initialFilterState);
  const visibleLessons = React.useMemo(() => {
    if (filter.level === 'All') return lessons;
    return lessons.filter((lesson) => lesson.level === filter.level);
  }, [filter.level, lessons]);
  const { progressLabel, reviewNext } = useLessonProgress(lessons.length);

  return (
    <main className="page-shell">
      <header className="page-shell__header">
        <p className="eyebrow">All React project roles</p>
        <h1>Template map</h1>
        <p>{progressLabel}</p>
      </header>

      <div className="filter-row">
        {['All', 'Core', 'Shared', 'Flow'].map((level) => (
          <button key={level} type="button" onClick={() => dispatch({ type: 'set-level', level })}>
            {level}
          </button>
        ))}
      </div>

      <button type="button" className="review-button" onClick={reviewNext}>
        Review one more template
      </button>

      <div className="card-grid">
        {visibleLessons.map((lesson) => (
          <LessonCard
            key={lesson.id}
            level={lesson.level}
            summary={lesson.summary}
            title={formatLesson(lesson.title)}
          />
        ))}
      </div>
    </main>
  );
}
```
