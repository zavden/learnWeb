## @file src/hooks/useToggle.js
## @lang javascript
## @role hook

```javascript
import React from 'react';

export function useToggle(initialValue = false) {
  const [enabled, setEnabled] = React.useState(initialValue);

  function toggle() {
    setEnabled((value) => !value);
  }

  return [enabled, toggle];
}
```
