import React from 'react';
import { initialState, sessionReducer } from './reducers/sessionReducer.js';
import { ActionList } from './components/ActionList.jsx';

export function App() {
  const [state, dispatch] = React.useReducer(sessionReducer, initialState);

  return (
    <main className="playground">
      <p className="eyebrow">Project JSX</p>
      <h1>Reducer project</h1>
      <ActionList count={state.completed} dispatch={dispatch} />
    </main>
  );
}
```

import React from 'react';

export function ActionList({ count, dispatch }) {
  return (
    <section className="card">
      <h2>Completed: {count}</h2>
      <div className="action-row">
        <button type="button" onClick={() => dispatch({ type: 'decrement' })}>-</button>
        <button type="button" onClick={() => dispatch({ type: 'increment' })}>+</button>
        <button type="button" onClick={() => dispatch({ type: 'reset' })}>Reset</button>
      </div>
    </section>
  );
}
```

## @file src/reducers/sessionReducer.js
## @lang javascript
## @role reducer

```javascript
export const initialState = {
  completed: 1,
};

export function sessionReducer(state, action) {
  if (action.type === 'increment') {
    return { ...state, completed: state.completed + 1 };
  }

  if (action.type === 'decrement') {
    return { ...state, completed: Math.max(0, state.completed - 1) };
  }

  if (action.type === 'reset') {
    return initialState;
  }

  return state;
}
```
