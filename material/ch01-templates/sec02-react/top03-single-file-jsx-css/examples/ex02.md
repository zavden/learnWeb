# JSX
```jsx
function App() {
  const alerts = ['Info', 'Success', 'Warning'];
  const [active, setActive] = React.useState(alerts[0]);

  return (
    <main className="alert-demo">
      <h1>{active}</h1>
      <div className="pill-row">
        {alerts.map((alert) => (
          <button key={alert} type="button" onClick={() => setActive(alert)}>
            {alert}
          </button>
        ))}
      </div>
    </main>
  );
}
```
