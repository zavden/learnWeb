# JSX
```jsx
function App() {
  const [liked, setLiked] = React.useState(false);

  return (
    <main className="card">
      <p className="eyebrow">JSX + CSS</p>
      <h1>Profile card</h1>
      <p>Style and interaction can still stay compact in one document.</p>
      <button type="button" onClick={() => setLiked(!liked)}>
        {liked ? 'Liked' : 'Like'}
      </button>
    </main>
  );
}
```
