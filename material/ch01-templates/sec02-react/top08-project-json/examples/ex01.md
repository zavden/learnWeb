## @file src/App.jsx
## @lang jsx
## @role app

```jsx
import React from 'react';
import dataset from './data/modules.json';
import { ModuleCard } from './components/ModuleCard.jsx';

export function App() {
  return (
    <main className="json-demo">
      <header className="hero">
        <p className="eyebrow">Project JSON</p>
        <h1>{dataset.title}</h1>
      </header>
      <div className="grid">
        {dataset.modules.map((module) => (
          <ModuleCard key={module.id} label={module.label} level={module.level} />
        ))}
      </div>
    </main>
  );
}
```
