## @file src/App.jsx
## @lang jsx
## @role app

```jsx
import React from 'react';
import plans from './data/plans.json';

export function App() {
  return (
    <main className="pricing">
      <p className="eyebrow">Project JSON</p>
      <h1>Pricing plans</h1>
      <div className="grid">
        {plans.map((plan) => (
          <article key={plan.id} className="plan-card">
            <h2>{plan.name}</h2>
            <strong>{plan.price}</strong>
            <p>{plan.summary}</p>
          </article>
        ))}
      </div>
    </main>
  );
}
```
