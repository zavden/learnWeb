# TSX
```tsx
type Action = { type: 'increment' } | { type: 'decrement' };

function reducer(score: number, action: Action): number {
  if (action.type === 'increment') return score + 1;
  return score - 1;
}

function App() {
  const [score, dispatch] = React.useReducer(reducer, 0);

  return (
    <main>
      <h1>Reducer score</h1>
      <p>{score}</p>
      <button type="button" onClick={() => dispatch({ type: 'decrement' })}>-</button>
      <button type="button" onClick={() => dispatch({ type: 'increment' })}>+</button>
    </main>
  );
}
```
