# JSX
```jsx
function App() {
  const colors = ['Sky', 'Mint', 'Amber'];
  const [selected, setSelected] = React.useState(colors[0]);

  return (
    <main>
      <h1>Choice Pills</h1>
      <p>Selected: {selected}</p>
      <div>
        {colors.map((color) => (
          <button key={color} type="button" onClick={() => setSelected(color)}>
            {color}
          </button>
        ))}
      </div>
    </main>
  );
}
```
