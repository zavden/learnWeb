# JSX
```jsx
function App() {
  const [count, setCount] = React.useState(0);

  return (
    <main>
      <h1>Counter</h1>
      <p>Single-file JSX keeps the focus on one component.</p>
      <button type="button" onClick={() => setCount(count + 1)}>
        Count: {count}
      </button>
    </main>
  );
}
```
