## @file src/pages/TemplatePage.tsx
## @lang tsx
## @role page

```tsx
import { useReducer } from 'react';
import { TemplateCard } from '../components/TemplateCard.tsx';
import { useWorkspaceSummary } from '../hooks/useWorkspaceSummary.ts';
import { useWorkspace } from '../context/WorkspaceContext.tsx';
import { formatTemplateName } from '../utils/formatTemplateName.ts';
import { initialTemplateState, templateReducer } from '../reducers/templateReducer.ts';

export function TemplatePage() {
  const { templates } = useWorkspace();
  const [state, dispatch] = useReducer(templateReducer, initialTemplateState);
  const summary = useWorkspaceSummary(templates, state.activeIndex);

  const activeTemplate = templates[state.activeIndex];

  return (
    <main className="screen">
      <p className="eyebrow">Project TSX + SCSS</p>
      <h1>{summary}</h1>
      <TemplateCard
        description={activeTemplate.description}
        title={formatTemplateName(activeTemplate.title)}
      />
      <div className="button-row">
        <button type="button" onClick={() => dispatch({ type: 'previous', size: templates.length })}>Previous</button>
        <button type="button" onClick={() => dispatch({ type: 'next', size: templates.length })}>Next</button>
      </div>
    </main>
  );
}
```

## @file src/context/WorkspaceContext.tsx
## @lang tsx
## @role context

```tsx
import { createContext, useContext, type ReactNode } from 'react';

type TemplateItem = {
  description: string;
  title: string;
};

type WorkspaceValue = {
  templates: TemplateItem[];
};

const templates: TemplateItem[] = [
  { title: 'page shell', description: 'A page groups the higher-level screen structure.' },
  { title: 'context provider', description: 'Context shares stable project data across the tree.' },
  { title: 'typed reducer', description: 'Reducers keep transitions explicit and predictable.' },
];

const WorkspaceContext = createContext<WorkspaceValue>({ templates });

export function WorkspaceProvider({ children }: { children: ReactNode }) {
  return (
    <WorkspaceContext.Provider value={{ templates }}>
      {children}
    </WorkspaceContext.Provider>
  );
}

export function useWorkspace() {
  return useContext(WorkspaceContext);
}
```

## @file src/hooks/useWorkspaceSummary.ts
## @lang typescript
## @role hook

```typescript
type TemplateItem = {
  description: string;
  title: string;
};

export function useWorkspaceSummary(templates: TemplateItem[], activeIndex: number): string {
  const active = templates[activeIndex];
  return `${activeIndex + 1}/${templates.length} · ${active.title}`;
}
```

## @file src/reducers/templateReducer.ts
## @lang typescript
## @role reducer

```typescript
export type TemplateState = {
  activeIndex: number;
};

export type TemplateAction =
  | { type: 'next'; size: number }
  | { type: 'previous'; size: number };

export const initialTemplateState: TemplateState = {
  activeIndex: 0,
};

export function templateReducer(state: TemplateState, action: TemplateAction): TemplateState {
  if (action.type === 'next') {
    return { activeIndex: (state.activeIndex + 1) % action.size };
  }

  return {
    activeIndex: (state.activeIndex - 1 + action.size) % action.size,
  };
}
```

## @file src/utils/formatTemplateName.ts
## @lang typescript
## @role util

```typescript
export function formatTemplateName(value: string): string {
  return value
    .split(' ')
    .map((part) => part.charAt(0).toUpperCase() + part.slice(1))
    .join(' ');
}
```
