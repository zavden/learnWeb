## @file src/App.tsx

## @role app

```tsx
import { StatRow } from './components/StatRow.tsx';
import { formatMetric } from './utils/formatMetric.ts';

const metrics = [
  { id: 'time', label: 'Build time', value: 12 },
  { id: 'lines', label: 'Useful lines', value: 84 },
  { id: 'tests', label: 'Checks', value: 6 },
];

export function App() {
  return (
    <main className="screen">
      <p className="eyebrow">Project TSX + SCSS</p>
      <h1>Util + component</h1>
      <div className="stat-grid">
        {metrics.map((metric) => (
          <StatRow key={metric.id} label={metric.label} value={formatMetric(metric.value)} />
        ))}
      </div>
    </main>
  );
}
```

## @file src/components/StatRow.tsx
## @lang tsx
## @role component

```tsx
type StatRowProps = {
  label: string;
  value: string;
};

export function StatRow({ label, value }: StatRowProps) {
  return (
    <article className="stat-row">
      <span>{label}</span>
      <strong>{value}</strong>
    </article>
  );
}
```

## @file src/utils/formatMetric.ts
## @lang typescript
## @role util

```typescript
export function formatMetric(value: number): string {
  return `${value.toString().padStart(2, '0')} pts`;
}
```
