## @file src/App.tsx
## @lang tsx
## @role app

```tsx
import { useState } from 'react';
import { SearchPanel } from './components/SearchPanel.tsx';
import { filterLessons, type Lesson } from './utils/filterLessons.ts';

const lessons: Lesson[] = [
  { id: 1, label: 'Props' },
  { id: 2, label: 'State' },
  { id: 3, label: 'Effects' },
  { id: 4, label: 'Reducers' },
];

export function App() {
  const [query, setQuery] = useState<string>('');
  const visibleLessons = filterLessons(lessons, query);

  return (
    <main className="dashboard">
      <p className="eyebrow">Project TSX + SCSS</p>
      <h1>Typed search</h1>
      <SearchPanel query={query} onQueryChange={setQuery} lessons={visibleLessons} />
    </main>
  );
}
```

## @file src/components/SearchPanel.tsx
## @lang tsx
## @role component

```tsx
import type { Lesson } from '../utils/filterLessons.ts';

type SearchPanelProps = {
  query: string;
  onQueryChange: (value: string) => void;
  lessons: Lesson[];
};

export function SearchPanel({ query, onQueryChange, lessons }: SearchPanelProps) {
  return (
    <section className="panel">
      <input
        type="text"
        value={query}
        placeholder="Search lessons"
        onChange={(event) => onQueryChange(event.target.value)}
      />
      <ul>
        {lessons.map((lesson) => <li key={lesson.id}>{lesson.label}</li>)}
      </ul>
    </section>
  );
}
```

## @file src/utils/filterLessons.ts
## @lang typescript
## @role util

```typescript
export type Lesson = {
  id: number;
  label: string;
};

export function filterLessons(lessons: Lesson[], query: string): Lesson[] {
  const normalizedQuery = query.trim().toLowerCase();
  if (!normalizedQuery) return lessons;

  return lessons.filter((lesson) => lesson.label.toLowerCase().includes(normalizedQuery));
}
```
