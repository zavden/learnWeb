import { VotePanel } from './components/VotePanel.tsx';
import { initialVoteState, voteReducer } from './reducers/voteReducer.ts';

export function App() {
  const [state, dispatch] = useReducer(voteReducer, initialVoteState);

  return (
    <main className="screen">
      <p className="eyebrow">Project TSX + SCSS</p>

}

## @file src/components/VotePanel.tsx
## @lang tsx
## @role component

```tsx
import type { VoteAction } from '../reducers/voteReducer.ts';

type VotePanelProps = {
  dispatch: React.Dispatch<VoteAction>;
  votes: number;
};

export function VotePanel({ dispatch, votes }: VotePanelProps) {
  return (
    <section className="panel">
      <h2>Votes: {votes}</h2>
      <div className="button-row">
        <button type="button" onClick={() => dispatch({ type: 'down' })}>Down</button>
        <button type="button" onClick={() => dispatch({ type: 'up' })}>Up</button>
        <button type="button" onClick={() => dispatch({ type: 'reset' })}>Reset</button>
      </div>
    </section>
  );
}
```

## @file src/reducers/voteReducer.ts
## @lang typescript
## @role reducer

```typescript
export type VoteState = {
  votes: number;
};

export type VoteAction =
  | { type: 'down' }
  | { type: 'up' }
  | { type: 'reset' };

export const initialVoteState: VoteState = {
  votes: 3,
};

export function voteReducer(state: VoteState, action: VoteAction): VoteState {
  if (action.type === 'up') {
    return { votes: state.votes + 1 };
  }

  if (action.type === 'down') {
    return { votes: Math.max(0, state.votes - 1) };
  }

  return initialVoteState;
}
```
