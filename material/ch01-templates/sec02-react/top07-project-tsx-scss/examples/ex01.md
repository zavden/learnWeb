## @file src/hooks/useCounter.ts
## @lang typescript
## @role hook

```typescript
import { useState } from 'react';

type CounterApi = {
  count: number;
  increment: () => void;
};

export function useCounter(initialValue: number): CounterApi {
  const [count, setCount] = useState<number>(initialValue);

  return {
    count,
    increment: () => setCount((value) => value + 1),
  };
}
```
