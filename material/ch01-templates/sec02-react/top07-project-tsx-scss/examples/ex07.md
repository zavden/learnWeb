## @file src/App.tsx

## @role app

```tsx
import { FilterChip } from './components/FilterChip.tsx';
import { useLessonFilter } from './hooks/useLessonFilter.ts';

export function App() {
  const {
    activeLevel,
    levels,
    setActiveLevel,
  } = useLessonFilter(['All', 'Core', 'Shared', 'Advanced']);

  return (
    <main className="screen">
      <p className="eyebrow">Project TSX + SCSS</p>
      <h1>Hook + typed component</h1>
      <div className="chip-row">
        {levels.map((level) => (
          <FilterChip
            key={level}
            active={level === activeLevel}
            label={level}
            onSelect={setActiveLevel}
          />
        ))}
      </div>
      <p>Selected level: {activeLevel}</p>
    </main>
  );
}
```

```tsx
type FilterChipProps = {
  active: boolean;
  label: string;
  onSelect: (value: string) => void;
};

export function FilterChip({ active, label, onSelect }: FilterChipProps) {
  return (
    <button
      type="button"
      className={active ? 'filter-chip filter-chip--active' : 'filter-chip'}
      onClick={() => onSelect(label)}
    >
      {label}
    </button>
  );
}
```

## @file src/hooks/useLessonFilter.ts
## @lang typescript
## @role hook

```typescript
import { useState } from 'react';

type LessonFilterApi = {
  activeLevel: string;
  levels: string[];
  setActiveLevel: (value: string) => void;
};

export function useLessonFilter(levels: string[]): LessonFilterApi {
  const [activeLevel, setActiveLevel] = useState<string>(levels[0] ?? 'All');

  return {
    activeLevel,
    levels,
    setActiveLevel,
  };
}
```
