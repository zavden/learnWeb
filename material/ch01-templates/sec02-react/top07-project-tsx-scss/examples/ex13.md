## @file src/context/FocusContext.tsx
## @lang tsx
## @role context

```tsx
import { createContext, useContext, useState, type ReactNode } from 'react';

type FocusContextValue = {
  focus: string;
  setFocus: (value: string) => void;
};

const FocusContext = createContext<FocusContextValue>({
  focus: 'Hooks',
  setFocus: () => {},
});

export function FocusProvider({ children }: { children: ReactNode }) {
  const [focus, setFocus] = useState<string>('Hooks');

  return (
    <FocusContext.Provider value={{ focus, setFocus }}>
      {children}
    </FocusContext.Provider>
  );
}

export function useFocusContext() {
  return useContext(FocusContext);
}
```

## @file src/hooks/useFocusText.ts
## @lang typescript
## @role hook

```typescript
import { useFocusContext } from '../context/FocusContext.tsx';

const lessons = ['Hooks', 'Reducers', 'Context'];

export function useFocusText() {
  const { focus, setFocus } = useFocusContext();

  function nextFocus() {
    const currentIndex = lessons.indexOf(focus);
    const nextIndex = (currentIndex + 1) % lessons.length;
    setFocus(lessons[nextIndex]);
  }

  return {
    focus,
    nextFocus,
  };
}
```
