# TSX
```tsx
type Tab = 'summary' | 'examples' | 'review';

function App() {
  const [activeTab, setActiveTab] = React.useState<Tab>('summary');

  return (
    <main className="tabs-card">
      <header>
        <p>TSX + CSS</p>
        <h1>{activeTab}</h1>
      </header>
      <div className="tab-row">
        <button type="button" onClick={() => setActiveTab('summary')}>Summary</button>
        <button type="button" onClick={() => setActiveTab('examples')}>Examples</button>
        <button type="button" onClick={() => setActiveTab('review')}>Review</button>
      </div>
    </main>
  );
}
```
