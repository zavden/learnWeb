## @file src/App.ts
## @lang typescript
## @role app

```typescript
import { defineComponent, ref } from 'vue';
import CounterCard from './components/CounterCard.ts';
import render from './App.html';

export default defineComponent({
  components: {
    CounterCard,
  },
  setup() {
    const title = ref<string>('Typed project shell');
    const count = ref<number>(2);
    const increment = () => {
      count.value += 1;
    };

    return {
      count,
      increment,
      title,
    };
  },
  render,
});
```

## @file src/components/CounterCard.ts
## @lang typescript
## @role component

```typescript
import { defineComponent, type PropType } from 'vue';
import render from './CounterCard.html';

export default defineComponent({
  emits: ['increment'],
  props: {
    label: {
      type: String as PropType<string>,
      required: true,
    },
    value: {
      type: Number as PropType<number>,
      required: true,
    },
  },
  render,
});
```
