## @file src/App.ts
## @lang typescript
## @role app

```typescript
import { defineComponent, ref } from 'vue';
import SummaryList from './components/SummaryList.ts';
import render from './App.html';

type SummaryItem = {
  id: number;
  label: string;
  value: number;
};

export default defineComponent({
  components: {
    SummaryList,
  },
  setup() {
    const title = ref<string>('Typed props');
    const items = ref<SummaryItem[]>([
      { id: 1, label: 'Refs', value: 2 },
      { id: 2, label: 'Computed', value: 1 },
      { id: 3, label: 'Components', value: 3 },
    ]);

    return {
      items,
      title,
    };
  },
  render,
});
```

import { defineComponent, type PropType } from 'vue';
import render from './SummaryList.html';

type SummaryItem = {

  label: string;
  value: number;
};

export default defineComponent({
  props: {

  render,
});
