## @file src/App.vue
## @lang vue
## @role app

```vue
<template>
  <main class="focus-shell">
    <div class="focus-shell__tabs">
      <button v-for="track in tracks" :key="track.id" type="button" @click="selectTrack(track.id)">
        {{ track.label }}
      </button>
    </div>
    <article class="focus-shell__panel">
      <h1>{{ current.label }}</h1>
      <p>{{ current.goal }}</p>
    </article>
  </main>
</template>

<script setup lang="ts">
import { computed, ref } from 'vue';

type Track = {
  id: string;
  label: string;
  goal: string;
};

const tracks: Track[] = [
  { id: 'state', label: 'State', goal: 'Model typed reactive values.' },
  { id: 'props', label: 'Props', goal: 'Pass data safely between components.' },
  { id: 'composables', label: 'Composables', goal: 'Share logic across files.' },
];

const currentId = ref<string>('state');
const current = computed<Track>(() => tracks.find((track) => track.id === currentId.value) ?? tracks[0]);
const selectTrack = (id: string) => {
  currentId.value = id;
};
</script>
```
