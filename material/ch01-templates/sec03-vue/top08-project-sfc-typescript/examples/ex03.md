## @file src/composables/useSprints.ts
## @lang typescript
## @role hook

```typescript
import { computed, ref } from 'vue';

type Sprint = {
  title: string;
  goal: string;
};

export function useSprints() {
  const items: Sprint[] = [
    { title: 'Sprint 1', goal: 'Type the component state.' },
    { title: 'Sprint 2', goal: 'Split the reusable pieces.' },
    { title: 'Sprint 3', goal: 'Compare the final result.' },
  ];
  const index = ref<number>(0);
  const current = computed<Sprint>(() => items[index.value]);
  const nextSprint = () => {
    index.value = (index.value + 1) % items.length;
  };

  return {
    current,
    nextSprint,
  };
}
```
