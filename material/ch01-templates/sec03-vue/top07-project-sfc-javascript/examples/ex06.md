## @file src/composables/useLessonProgress.js
## @lang javascript
## @role hook

```javascript
import { computed } from 'vue';

export function useLessonProgress(reviewedCount, filteredLessons) {
  const progressLabel = computed(() => `${reviewedCount.value}/${filteredLessons.value.length} templates reviewed`);

  function reviewNext() {
    reviewedCount.value = Math.min(filteredLessons.value.length, reviewedCount.value + 1);
  }

  function resetProgress() {
    reviewedCount.value = 0;
  }

  return {
    progressLabel,
    resetProgress,
    reviewNext,
  };
}
```

## @file src/stores/useLessonStore.js
## @lang javascript
## @role context

```javascript
import { computed, ref } from 'vue';

const lessons = ref([
  { id: 'component', title: 'component card', level: 'Core', summary: 'Reusable UI lives in child SFC files.' },
  { id: 'composable', title: 'composable logic', level: 'Shared', summary: 'Composables reuse Composition API logic.' },
  { id: 'page', title: 'page shell', level: 'Flow', summary: 'Pages compose the smaller project files.' },
]);

const currentLevel = ref('All');

export function useLessonStore() {
  const levels = computed(() => ['All', ...new Set(lessons.value.map((lesson) => lesson.level))]);
  const filteredLessons = computed(() => {
    if (currentLevel.value === 'All') return lessons.value;
    return lessons.value.filter((lesson) => lesson.level === currentLevel.value);
  });

  function setLevel(level) {
    currentLevel.value = level;
  }

  return {
    currentLevel,
    filteredLessons,
    lessons,
    levels,
    setLevel,
  };
}
```

## @file src/composables/useLessonSummary.js
## @lang javascript
## @role reducer

```javascript
import { computed } from 'vue';

export function useLessonSummary(filteredLessons, reviewedCount) {
  return computed(() => {
    if (!filteredLessons.value.length) {
      return 'No lessons match the current filter.';
    }

    const currentLesson = filteredLessons.value[Math.min(reviewedCount.value, filteredLessons.value.length - 1)];
    return `Current focus: ${currentLesson.title}`;
  });
}
```

## @file src/utils/formatLesson.js
## @lang javascript
## @role util

```javascript
export function formatLesson(value) {
  return value
    .split(' ')
    .map((part) => part.charAt(0).toUpperCase() + part.slice(1))
    .join(' ');
}
```
