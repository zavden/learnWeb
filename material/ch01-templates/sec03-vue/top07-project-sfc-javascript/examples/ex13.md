## @file src/stores/useInboxStore.js
## @lang javascript
## @role context

```javascript
import { computed, ref } from 'vue';

const messages = ref([
  { id: 1, sender: 'Aya', subject: 'Review the component map', unread: true },
  { id: 2, sender: 'Leo', subject: 'Page shell is ready', unread: false },
  { id: 3, sender: 'Mina', subject: 'Store helpers need a label', unread: true },
]);

const filter = ref('All');

export function useInboxStore() {
  const filteredMessages = computed(() => {
    if (filter.value === 'Unread') {
      return messages.value.filter((message) => message.unread);
    }

    return messages.value;
  });

  function setFilter(value) {
    filter.value = value;
  }

  return {
    filter,
    filteredMessages,
    messages,
    setFilter,
  };
}
```

## @file src/composables/useInboxSummary.js
## @lang javascript
## @role reducer

```javascript
import { computed } from 'vue';

export function useInboxSummary(filteredMessages, filter) {
  return computed(() => `${filteredMessages.value.length} messages shown in ${filter.value} mode.`);
}
```
