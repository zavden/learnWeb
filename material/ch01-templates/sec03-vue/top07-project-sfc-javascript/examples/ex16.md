## @file src/composables/usePaletteStepper.js
## @lang javascript
## @role hook

```javascript
import { computed } from 'vue';

export function usePaletteStepper(palettes, selectedIndex) {
  const visiblePalette = computed(() => palettes.value[selectedIndex.value]);

  function nextPalette() {
    selectedIndex.value = (selectedIndex.value + 1) % palettes.value.length;
  }

  return {
    nextPalette,
    visiblePalette,
  };
}
```

export function useSelectionSummary(visiblePalette, selectedIndex) {
  return computed(() => `Palette ${selectedIndex.value + 1} with ${visiblePalette.value.length} swatches`);
}

## @file src/utils/formatSwatchName.js
## @lang javascript
## @role util

```javascript
export function formatSwatchName(value) {
  return value
    .split(' ')
    .map((part) => part.charAt(0).toUpperCase() + part.slice(1))
    .join(' ');
}
```
