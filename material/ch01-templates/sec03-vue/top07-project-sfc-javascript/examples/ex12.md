```vue
<template>
  <main class="board-shell">
    <h1>Store combination</h1>
    <p>Current column: {{ currentColumn }}</p>
    <div class="board-actions">
      <button v-for="column in columns" :key="column" type="button" @click="setColumn(column)">
        {{ column }}
      </button>
    </div>
  </main>
</template>

<script setup>

const { columns, currentColumn, setColumn } = useBoardStore();
</script>
```

## @file src/stores/useBoardStore.js
## @lang javascript
## @role context

```javascript
import { ref } from 'vue';

const columns = ['Backlog', 'Doing', 'Done'];
const currentColumn = ref('Backlog');

export function useBoardStore() {
  function setColumn(column) {
    currentColumn.value = column;
  }

  return {
    columns,
    currentColumn,
    setColumn,
  };
}
```
