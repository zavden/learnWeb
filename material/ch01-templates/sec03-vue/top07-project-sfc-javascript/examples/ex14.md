## @file src/composables/useTipRotation.js
## @lang javascript
## @role hook

```javascript
import { computed, ref } from 'vue';

export function useTipRotation() {
  const tips = [
    { title: 'Page first', detail: 'Let a page SFC own the overall screen structure.' },
    { title: 'Component next', detail: 'Move repeated cards into a child SFC.' },
    { title: 'Composable last', detail: 'Extract reusable logic when state starts repeating.' },
  ];

  const index = ref(0);
  const currentTip = computed(() => tips[index.value]);

  function nextTip() {
    index.value = (index.value + 1) % tips.length;
  }

  return {
    currentTip,
    nextTip,
  };
}
```
