## @file src/composables/useLessonSteps.js
## @lang javascript
## @role hook

```javascript
import { computed, ref } from 'vue';

export function useLessonSteps() {
  const steps = [
    { title: 'Entry file', detail: 'Boot the app with createApp().' },
    { title: 'Root SFC', detail: 'Use App.vue as the top component.' },
    { title: 'Child files', detail: 'Move repeated UI into more SFC files.' },
  ];

  const index = ref(0);
  const current = computed(() => steps[index.value]);

  function nextStep() {
    index.value = (index.value + 1) % steps.length;
  }

  return {
    current,
    nextStep,
  };
}
```
