## @file src/App.js
## @lang javascript
## @role app

```javascript
import { defineComponent, ref } from 'vue';
import ResourceList from './components/ResourceList.js';
import render from './App.html';

export default defineComponent({
  components: {
    ResourceList,
  },
  setup() {
    const title = ref('Props across files');
    const resources = ref([
      { id: 1, label: 'Template syntax', level: 'Easy' },
      { id: 2, label: 'Events', level: 'Medium' },
      { id: 3, label: 'Composables', level: 'Medium' },
    ]);

    return {
      resources,
      title,
    };
  },
  render,
});
```
