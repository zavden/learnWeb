## @file src/App.js
## @lang javascript
## @role app

```javascript
import { computed, defineComponent, ref } from 'vue';
import ProgressCard from './components/ProgressCard.js';
import render from './App.html';

export default defineComponent({
  components: {
    ProgressCard,
  },
  setup() {
    const completed = ref(1);
    const total = ref(4);
    const summary = computed(() => `${completed.value}/${total.value} lessons complete.`);
    const completeLesson = () => {
      if (completed.value < total.value) {
        completed.value += 1;
      }
    };

    return {
      completeLesson,
      completed,
      summary,
      total,
    };
  },
  render,
});
```
