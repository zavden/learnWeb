## @file src/App.js
## @lang javascript
## @role app

```javascript
import { defineComponent, ref } from 'vue';
import LessonCard from './components/LessonCard.js';
import render from './App.html';

export default defineComponent({
  components: {
    LessonCard,
  },
  setup() {
    const title = ref('Vue project JavaScript');
    const lessons = ref([
      'Split render files',
      'Reusable components',
      'Shared stylesheet',
    ]);

    return {
      lessons,
      title,
    };
  },
  render,
});
```
