# TypeScript

```typescript
import { defineComponent, ref } from 'vue';

type Theme = {
  description: string;
};

export default defineComponent({
  setup() {
    const title = ref<string>('Optional CSS still works');
    const theme = ref<Theme>({
      description: 'Block-based TypeScript can still carry some lightweight styling.',
    });
    const switchTheme = () => {
      theme.value = {
        description: 'The same typed component can swap content reactively.',
      };
    };

    return {
      switchTheme,
      theme,
      title,
    };
  },
});
```
