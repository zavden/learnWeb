# HTML

```html
<section>
  <h1>{{ title }}</h1>
  <p>Count: {{ count }}</p>
  <button type="button" @click="increment">Increment</button>
</section>
```

# TypeScript

```typescript
import { defineComponent, ref } from 'vue';

export default defineComponent({
  setup() {
    const title = ref<string>('Typed counter');
    const count = ref<number>(0);
    const increment = () => {
      count.value += 1;
    };

    return {
      count,
      increment,
      title,
    };
  },
});
```
