# HTML

```html
<section>
  <h1>{{ title }}</h1>
  <input v-model="draft" type="text" placeholder="New lesson" />
  <button type="button" @click="addLesson">Add lesson</button>
  <ul>
    <li v-for="lesson in lessons" :key="lesson">{{ lesson }}</li>
  </ul>
</section>
```

# JavaScript

```javascript
import { ref } from 'vue';

export default {
  setup() {
    const title = ref('Reactive lists');
    const draft = ref('');
    const lessons = ref(['Template syntax', 'Refs']);
    const addLesson = () => {
      const value = draft.value.trim();
      if (!value) {
        return;
      }
      lessons.value = [...lessons.value, value];
      draft.value = '';
    };

    return {
      addLesson,
      draft,
      lessons,
      title,
    };
  },
};
```
