---
framework: vue

# HTML

```html
<section>
  <h1>{{ title }}</h1>
  <p>Count: {{ count }}</p>
  <button type="button" @click="increment">Increment</button>
</section>
```

# JavaScript

```javascript
import { ref } from 'vue';

export default {
  setup() {
    const title = ref('Counter with refs');
    const count = ref(0);
    const increment = () => {
      count.value += 1;
    };

    return {
      count,
      increment,
      title,
    };
  },
};
```
