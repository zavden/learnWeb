# JavaScript

```javascript
import { computed, ref } from 'vue';

export default {
  setup() {
    const title = ref('Progress tracker');
    const completed = ref(2);
    const total = ref(5);
    const summary = computed(() => `Completed ${completed.value} of ${total.value} lessons.`);
    const complete = () => {
      if (completed.value < total.value) {
        completed.value += 1;
      }
    };

    return {
      complete,
      summary,
      title,
    };
  },
};
```
