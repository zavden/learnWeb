# HTML

```html
<section class="palette">
  <h1>{{ title }}</h1>
  <p>{{ active.swatch }}</p>
  <div class="palette__chips">
    <button
      v-for="option in options"
      :key="option.id"
      type="button"
      class="palette__chip"
      @click="selectOption(option.id)"
    >
      {{ option.label }}
    </button>
  </div>
</section>
```

# JavaScript

```javascript
import { computed, ref } from 'vue';

export default {
  setup() {
    const title = ref('Reactive palette');
    const options = ref([
      { id: 'ocean', label: 'Ocean', swatch: '#0ea5e9 / #082f49' },
      { id: 'lime', label: 'Lime', swatch: '#84cc16 / #365314' },
      { id: 'rose', label: 'Rose', swatch: '#fb7185 / #881337' },
    ]);
    const activeId = ref('ocean');
    const active = computed(() => (
      options.value.find((option) => option.id === activeId.value) ?? options.value[0]
    ));
    const selectOption = (id) => {
      activeId.value = id;
    };

    return {
      active,
      options,
      selectOption,
      title,
    };
  },
};
```
