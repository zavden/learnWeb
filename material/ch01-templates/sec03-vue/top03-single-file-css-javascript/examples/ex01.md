# HTML

```html
<section class="card">
  <p class="eyebrow">Vue + CSS</p>
  <h1>{{ title }}</h1>
  <p>{{ message }}</p>
  <button type="button" @click="toggleMood">{{ buttonLabel }}</button>
</section>
```

# JavaScript

```javascript
import { computed, ref } from 'vue';

export default {
  setup() {
    const title = ref('Styled card');
    const happy = ref(true);
    const message = computed(() => (
      happy.value ? 'CSS makes small Vue examples easier to scan.' : 'The same reactive template now uses a calm message.'
    ));
    const buttonLabel = computed(() => (happy.value ? 'Use calm copy' : 'Use bright copy'));
    const toggleMood = () => {
      happy.value = !happy.value;
    };

    return {
      buttonLabel,
      message,
      title,
      toggleMood,
    };
  },
};
```
