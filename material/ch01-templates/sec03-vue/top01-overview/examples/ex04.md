```javascript
import App from './App.js';

createApp(App).mount('#app');
```

## @file src/App.js
## @lang javascript
## @role app

```javascript
import { defineComponent, ref } from 'vue';
import TopicCard from './components/TopicCard.js';
import render from './App.html';

export default defineComponent({
  components: {
    TopicCard,
  },
  setup() {
    const topics = ref(['Templates', 'Components', 'Composables']);

    return {
      topics,
    };
  },
  render,
});
```

## @file src/App.html
## @lang html
## @role markup

```html
<main class="overview-grid">
  <h1>Vue project mode</h1>
  <TopicCard v-for="topic in topics" :key="topic" :title="topic" />
</main>
```
