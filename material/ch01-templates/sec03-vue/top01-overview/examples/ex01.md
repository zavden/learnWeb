# HTML

```html
<main>
  <p>Vue overview</p>
  <h1>{{ title }}</h1>
  <p>{{ summary }}</p>
  <button type="button" @click="advance">Advance lesson</button>
</main>
```

# JavaScript

```javascript
import { computed, ref } from 'vue';

export default {
  setup() {
    const title = ref('Single-file JavaScript');
    const lessons = ['Refs', 'Computed', 'Events'];
    const index = ref(0);
    const summary = computed(() => `Now studying: ${lessons[index.value]}`);
    const advance = () => {
      index.value = (index.value + 1) % lessons.length;
    };

    return {
      advance,
      summary,
      title,
    };
  },
};
```
