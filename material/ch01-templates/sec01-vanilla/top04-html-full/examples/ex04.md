```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0" />
  <title>Script Order</title>
</head>
<body>
  <main>
    <h1 id="headline">Loading order</h1>
  </main>
  <script>
    const headline = document.getElementById('headline');
    if (headline) {
      headline.textContent = 'Inline script ran after the body was parsed';
    }
  </script>
</body>
</html>
```
