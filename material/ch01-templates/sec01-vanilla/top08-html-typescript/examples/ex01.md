# HTML

```html
<button id="step-button" type="button">Advance step</button>
<p id="step-output">Step 1</p>
```

# TypeScript

```typescript
let step: number = 1;
const button = document.getElementById('step-button') as HTMLButtonElement | null;
const output = document.getElementById('step-output') as HTMLParagraphElement | null;

button?.addEventListener('click', () => {
  step += 1;
  if (output) {
    output.textContent = `Step ${step}`;
  }
});
```
