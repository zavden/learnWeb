# HTML

```html
<label>
  Progress
  <input id="progress-range" type="range" min="0" max="100" value="25" />
</label>
<p id="progress-text">25%</p>
```

# TypeScript

```typescript
const range = document.getElementById('progress-range') as HTMLInputElement | null;
const text = document.getElementById('progress-text') as HTMLParagraphElement | null;

range?.addEventListener('input', () => {
  if (text) {
    text.textContent = `${range.value}%`;
  }
});
```
