```html
  <h1>Theme Switch</h1>
  <button id="toggle-theme" type="button">Switch</button>
```


# JavaScript

```javascript
const button = document.getElementById('toggle-theme');

button?.addEventListener('click', () => {
  document.body.classList.toggle('dark');
});
```
