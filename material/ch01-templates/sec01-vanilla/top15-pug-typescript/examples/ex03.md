# Pug

```pug
form#task-form
  input#task-input(type="text" placeholder="Write a task")
  button(type="submit") Save
p#task-status Waiting...
```

# TypeScript

```typescript
const form = document.getElementById('task-form') as HTMLFormElement | null;
const input = document.getElementById('task-input') as HTMLInputElement | null;
const status = document.getElementById('task-status') as HTMLParagraphElement | null;

form?.addEventListener('submit', (event: SubmitEvent) => {
  event.preventDefault();
  const value = input?.value.trim() || 'Untitled task';
  if (status) {
    status.textContent = `Saved: ${value}`;
  }
  console.log('saved-task', value);
});
```
