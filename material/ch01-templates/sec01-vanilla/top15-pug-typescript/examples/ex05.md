# Pug

```pug
button#status-btn(type="button") Set online
p#status-text Offline
```

# TypeScript

```typescript
let online: boolean = false;
const button = document.getElementById('status-btn') as HTMLButtonElement | null;
const text = document.getElementById('status-text') as HTMLParagraphElement | null;

button?.addEventListener('click', () => {
  online = !online;
  if (button && text) {
    button.textContent = online ? 'Set offline' : 'Set online';
    text.textContent = online ? 'Online' : 'Offline';
  }
});
```
