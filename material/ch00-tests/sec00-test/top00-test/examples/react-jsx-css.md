# JSX
```jsx
function App() {
  const [active, setActive] = React.useState(false);

  return (
    <section className={`card ${active ? 'is-active' : ''}`}>
      <p className="eyebrow">React single-file</p>
      <h1>Styled JSX example</h1>
      <p>The HTML shell is generated by the platform.</p>
      <button type="button" onClick={() => setActive(!active)}>
        {active ? 'Deactivate' : 'Activate'}
      </button>
    </section>
  );
}
```
