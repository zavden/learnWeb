---
renderer: shader
resolution: 960x540
shader_uniforms: ink:vec4=0.08,0.86,0.72,0.82|paper:vec3=0.97,0.94,0.86
---

# Vertex

```vertex
attribute vec2 a_position;
varying vec2 v_uv;

void main() {
  v_uv = a_position * 0.5 + 0.5;
  gl_Position = vec4(a_position, 0.0, 1.0);
}
```

# Fragment

```fragment
precision mediump float;

uniform vec2 u_resolution;
uniform vec4 ink;
uniform vec3 paper;

void main() {
  vec2 uv = gl_FragCoord.xy / u_resolution;
  float stripe = smoothstep(0.35, 0.65, 0.5 + 0.5 * sin(uv.x * 18.0));
  float grain = 0.5 + 0.5 * sin((uv.x + uv.y) * 42.0);
  vec3 overlay = mix(paper, ink.rgb, stripe * ink.a);
  vec3 color = mix(overlay, ink.rgb, grain * ink.a * 0.28);
  gl_FragColor = vec4(color, 1.0);
}
```
