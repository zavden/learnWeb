---
renderer: shader
resolution: 800x800
---

# Vertex

```vertex
attribute vec2 a_position;
varying vec2 v_uv;

void main() {
  v_uv = a_position * 0.5 + 0.5;
  gl_Position = vec4(a_position, 0.0, 1.0);
}
```

# Fragment

```fragment
precision mediump float;

uniform vec2 u_mouse;
uniform float u_mouse_pressed;
uniform vec2 u_resolution;

void main() {
  vec2 uv = gl_FragCoord.xy / u_resolution;
  vec2 mouse = u_mouse / u_resolution;
  float distanceToMouse = distance(uv, mouse);
  float ring = smoothstep(0.24, 0.16, distanceToMouse);
  float core = smoothstep(0.08, 0.02, distanceToMouse);
  vec3 base = vec3(0.04, 0.06, 0.12) + vec3(uv.x * 0.16, uv.y * 0.18, 0.12);
  vec3 accent = mix(vec3(0.28, 0.82, 1.0), vec3(1.0, 0.46, 0.28), u_mouse_pressed);
  vec3 color = base + accent * ring + vec3(1.0) * core;
  gl_FragColor = vec4(color, 1.0);
}
```
