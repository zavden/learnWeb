# Vertex

```vertex
attribute vec2 a_position;
varying vec2 v_uv;

void main() {
  v_uv = a_position * 0.5 + 0.5;
  gl_Position = vec4(a_position, 0.0, 1.0);
}
```

# Fragment

```fragment
precision mediump float;

uniform vec2 u_resolution;
uniform vec3 base;
uniform vec3 glow;

void main() {
  vec2 uv = gl_FragCoord.xy / u_resolution;
  vec2 centered = uv - 0.5;
  float radius = length(centered * vec2(1.4, 1.0));
  float pulse = smoothstep(0.58, 0.08, radius);
  vec3 color = mix(base, glow, pulse);
  color += vec3(uv.x * 0.08, uv.y * 0.06, 0.02);
  gl_FragColor = vec4(color, 1.0);
}
```
