# Vertex

```vertex
attribute vec2 a_position;
varying vec2 v_uv;

void main() {
  v_uv = a_position * 0.5 + 0.5;
  gl_Position = vec4(a_position, 0.0, 1.0);
}
```

# Fragment

```fragment
precision mediump float;

uniform vec2 u_resolution;
uniform float intensity;
uniform bool invert;
uniform vec2 focus;

void main() {
  vec2 uv = gl_FragCoord.xy / u_resolution;
  float distanceToFocus = distance(uv, focus);
  float halo = smoothstep(0.55, 0.12, distanceToFocus);
  vec3 color = vec3(uv.x * intensity, uv.y * intensity, halo * intensity);

  if (invert) {
    color = 1.0 - color;
  }

  gl_FragColor = vec4(color, 1.0);
}
```
