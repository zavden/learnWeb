---
renderer: shader
resolution: 960x540
shader_textures: u_checker=checker.svg|u_spot=spotlight.svg
---

# Vertex

```vertex
attribute vec2 a_position;
varying vec2 v_uv;

void main() {
  v_uv = a_position * 0.5 + 0.5;
  gl_Position = vec4(a_position, 0.0, 1.0);
}
```

# Fragment

```fragment
precision mediump float;

uniform vec2 u_resolution;
uniform sampler2D u_checker;
uniform sampler2D u_spot;

void main() {
  vec2 uv = gl_FragCoord.xy / u_resolution;
  vec3 checker = texture2D(u_checker, uv * 2.0).rgb;
  vec3 spot = texture2D(u_spot, uv).rgb;
  vec3 color = mix(checker, spot, 0.45);
  gl_FragColor = vec4(color, 1.0);
}
```
