---
renderer: shader
resolution: 1024x576
---

# Vertex

```vertex
attribute vec2 a_position;
varying vec2 v_uv;

void main() {
  v_uv = a_position * 0.5 + 0.5;
  gl_Position = vec4(a_position, 0.0, 1.0);
}
```

# Fragment

```fragment
precision mediump float;

uniform float u_frame;
uniform vec2 u_resolution;

void main() {
  vec2 uv = gl_FragCoord.xy / u_resolution;
  float stepped = mod(floor(u_frame / 6.0), 18.0) / 18.0;
  float stripes = step(0.5, fract(uv.y * 10.0 + stepped * 8.0));
  vec3 color = mix(vec3(0.08, 0.11, 0.18), vec3(0.95, 0.62, 0.18), stripes);
  color += vec3(stepped * 0.35, uv.x * 0.18, 0.05);
  gl_FragColor = vec4(color, 1.0);
}
```
