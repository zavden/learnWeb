# Vertex

```vertex
attribute vec2 a_position;
varying vec2 v_uv;

void main() {
  v_uv = a_position * 0.5 + 0.5;
  gl_Position = vec4(a_position, 0.0, 1.0);
}
```

# Fragment

```fragment
precision mediump float;

uniform vec2 u_resolution;
uniform float intensity;
uniform int test;
int bands = test;
void main() {
  vec2 uv = gl_FragCoord.xy / u_resolution;
  float bandCount = max(float(bands), 1.0);
  float stepped = floor(uv.x * bandCount) / bandCount;
  vec3 color = vec3(stepped * intensity, uv.y * intensity, 1.0 - stepped * 0.5);
  gl_FragColor = vec4(color, 1.0);
}
```
