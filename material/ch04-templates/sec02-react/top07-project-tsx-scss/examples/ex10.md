## @file src/hooks/useTopicRotation.ts
## @lang typescript
## @role hook

```typescript
import { useState } from 'react';

type TopicRotationApi = {
  currentTopic: string;
  nextTopic: () => void;
};

export function useTopicRotation(topics: string[]): TopicRotationApi {
  const [index, setIndex] = useState<number>(0);

  return {
    currentTopic: topics[index] ?? '',
    nextTopic: () => setIndex((value) => (value + 1) % topics.length),
  };
}
```
