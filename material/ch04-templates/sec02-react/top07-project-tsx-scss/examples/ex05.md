## @file src/hooks/useStepper.ts
## @lang typescript
## @role hook

```typescript
import { useState } from 'react';

type StepperApi = {
  activeStep: string;
  nextStep: () => void;
  previousStep: () => void;
};

export function useStepper(steps: string[]): StepperApi {
  const [index, setIndex] = useState<number>(0);

  function nextStep() {
    setIndex((value) => Math.min(value + 1, steps.length - 1));
  }

  function previousStep() {
    setIndex((value) => Math.max(value - 1, 0));
  }

  return {
    activeStep: steps[index],
    nextStep,
    previousStep,
  };
}
```
