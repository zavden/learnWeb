import { useReducer } from 'react';
import { SprintCounter } from '../components/SprintCounter.tsx';
import { initialSprintState, sprintReducer } from '../reducers/sprintReducer.ts';

export function SprintPage() {
  const [state, dispatch] = useReducer(sprintReducer, initialSprintState);

  return (
    <main className="screen">
      <p className="eyebrow">Project TSX + SCSS</p>
      <h1>Page + reducer</h1>
      <SprintCounter dispatch={dispatch} total={state.total} />
    </main>
  );
}

## @file src/components/SprintCounter.tsx
## @lang tsx
## @role component

```tsx
import type { SprintAction } from '../reducers/sprintReducer.ts';

type SprintCounterProps = {
  dispatch: React.Dispatch<SprintAction>;
  total: number;
};

export function SprintCounter({ dispatch, total }: SprintCounterProps) {
  return (
    <section className="panel">
      <h2>Stories delivered: {total}</h2>
      <div className="button-row">
        <button type="button" onClick={() => dispatch({ type: 'add' })}>Add one</button>
        <button type="button" onClick={() => dispatch({ type: 'clear' })}>Clear</button>
      </div>
    </section>
  );
}
```

## @file src/reducers/sprintReducer.ts
## @lang typescript
## @role reducer

```typescript
export type SprintState = {
  total: number;
};

export type SprintAction =
  | { type: 'add' }
  | { type: 'clear' };

export const initialSprintState: SprintState = {
  total: 2,
};

export function sprintReducer(state: SprintState, action: SprintAction): SprintState {
  if (action.type === 'add') {
    return { total: state.total + 1 };
  }

  return initialSprintState;
}
```
