```tsx
import type { BoardAction } from '../reducers/boardReducer.ts';

type BoardSummaryProps = {
  completed: number;
  dispatch: React.Dispatch<BoardAction>;
};

export function BoardSummary({ completed, dispatch }: BoardSummaryProps) {
  return (
    <section className="panel">
      <h2>Completed tasks: {completed}</h2>
      <div className="button-row">
        <button type="button" onClick={() => dispatch({ type: 'decrement' })}>-</button>
        <button type="button" onClick={() => dispatch({ type: 'increment' })}>+</button>
        <button type="button" onClick={() => dispatch({ type: 'reset' })}>Reset</button>
      </div>
    </section>
  );
}
```

## @file src/reducers/boardReducer.ts
## @lang typescript
## @role reducer

```typescript
export type BoardState = {
  completed: number;
};

export type BoardAction =
  | { type: 'increment' }
  | { type: 'decrement' }
  | { type: 'reset' };

export const initialBoardState: BoardState = {
  completed: 1,
};

export function boardReducer(state: BoardState, action: BoardAction): BoardState {
  if (action.type === 'increment') {
    return { ...state, completed: state.completed + 1 };
  }

  if (action.type === 'decrement') {
    return { ...state, completed: Math.max(0, state.completed - 1) };
  }

  if (action.type === 'reset') {
    return initialBoardState;
  }

  return state;
}
```
