## @role page

```tsx
import { AgendaRow } from '../components/AgendaRow.tsx';
import { formatDuration } from '../utils/formatDuration.ts';

const agenda = [
  { id: 'entry', label: 'Bootstrap the root', minutes: 10 },
  { id: 'page', label: 'Compose the page shell', minutes: 20 },
  { id: 'util', label: 'Format display values', minutes: 5 },
];

export function AgendaPage() {
  return (
    <main className="screen">
      <p className="eyebrow">Project TSX + SCSS</p>
      <h1>Page + util</h1>
      <div className="agenda-grid">
        {agenda.map((item) => (
          <AgendaRow key={item.id} label={item.label} time={formatDuration(item.minutes)} />
        ))}
      </div>
    </main>
  );
}
```
