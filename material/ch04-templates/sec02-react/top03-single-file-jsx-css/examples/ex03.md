# JSX
```jsx
function App() {
  const [progress, setProgress] = React.useState(30);

  return (
    <main className="progress-card">
      <h1>Learning progress</h1>
      <div className="meter">
        <span className="meter-fill" style={{ width: `${progress}%` }} />
      </div>
      <p>{progress}% complete</p>
      <button type="button" onClick={() => setProgress((value) => Math.min(value + 10, 100))}>
        Advance
      </button>
    </main>
  );
}
```
