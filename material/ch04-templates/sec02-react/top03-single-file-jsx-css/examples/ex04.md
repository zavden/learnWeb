# JSX
```jsx
function App() {
  const quotes = [
    'Components help split the interface.',
    'State makes the UI react to events.',
    'Props move data across the tree.',
  ];
  const [index, setIndex] = React.useState(0);

  return (
    <main className="quote-card">
      <p className="eyebrow">React quote</p>
      <blockquote>{quotes[index]}</blockquote>
      <button type="button" onClick={() => setIndex((index + 1) % quotes.length)}>
        Next quote
      </button>
    </main>
  );
}
```
