# JSX
```jsx
function App() {
  const items = [
    { id: 1, question: 'What is a component?', answer: 'A reusable UI function.' },
    { id: 2, question: 'What is state?', answer: 'Data that changes over time.' },
    { id: 3, question: 'Why use JSX?', answer: 'It lets markup live close to logic.' },
  ];
  const [openId, setOpenId] = React.useState(1);

  return (
    <main>
      <h1>FAQ</h1>
      {items.map((item) => (
        <section key={item.id}>
          <button type="button" onClick={() => setOpenId(item.id)}>
            {item.question}
          </button>
          {openId === item.id ? <p>{item.answer}</p> : null}
        </section>
      ))}
    </main>
  );
}
```
