# JSX
```jsx
function App() {
  const tabs = ['Overview', 'Hooks', 'Projects'];
  const [activeTab, setActiveTab] = React.useState(tabs[0]);

  React.useEffect(() => {
    console.info('active-tab', activeTab);
  }, [activeTab]);

  return (
    <main>
      <h1>{activeTab}</h1>
      <p>Open the console to inspect effect order.</p>
      <div>
        {tabs.map((tab) => (
          <button key={tab} type="button" onClick={() => setActiveTab(tab)}>
            {tab}
          </button>
        ))}
      </div>
    </main>
  );
}
```
