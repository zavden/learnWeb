## @role app

```jsx
import React from 'react';
import { LessonCard } from './components/LessonCard.jsx';

const lessons = [
  { id: 1, title: 'Props', summary: 'Pass data down the tree.' },
  { id: 2, title: 'State', summary: 'Keep local UI data reactive.' },
  { id: 3, title: 'Effects', summary: 'Run side effects after render.' },
];

export function App() {
  return (
    <main className="gallery">
      <header className="hero">
        <p className="eyebrow">Project JSX</p>
        <h1>Props gallery</h1>
      </header>
      <div className="card-grid">
        {lessons.map((lesson) => (
          <LessonCard key={lesson.id} title={lesson.title} summary={lesson.summary} />
        ))}
      </div>
    </main>
  );
}
```
