# TSX
```tsx
function App() {
  const inputRef = React.useRef<HTMLInputElement | null>(null);
  const [value, setValue] = React.useState<string>('React');

  return (
    <main>
      <h1>Typed ref</h1>
      <input
        ref={inputRef}
        type="text"
        value={value}
        onChange={(event) => setValue(event.target.value)}
      />
      <button type="button" onClick={() => inputRef.current?.focus()}>
        Focus input
      </button>
    </main>
  );
}
```
