# TSX
```tsx
type Flashcard = {
  id: number;
  prompt: string;
  answer: string;
};

const cards: Flashcard[] = [
  { id: 1, prompt: 'What is JSX?', answer: 'A syntax extension for UI markup.' },
  { id: 2, prompt: 'What is state?', answer: 'Component data that changes.' },
  { id: 3, prompt: 'What is a prop?', answer: 'Input passed to a component.' },
];

function App() {
  const [index, setIndex] = React.useState<number>(0);
  const card = cards[index];

  return (
    <main>
      <h1>{card.prompt}</h1>
      <p>{card.answer}</p>
      <button type="button" onClick={() => setIndex((index + 1) % cards.length)}>
        Next card
      </button>
    </main>
  );
}
```
