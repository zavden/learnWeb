## @file src/App.jsx
## @lang jsx
## @role app

```jsx
import React from 'react';
import dataset from './data/faqs.json';

export function App() {
  const [activeId, setActiveId] = React.useState(dataset.items[0].id);

  return (
    <main className="faq-demo">
      <p className="eyebrow">Project JSON</p>
      <h1>{dataset.title}</h1>
      <div className="faq-list">
        {dataset.items.map((item) => (
          <section key={item.id} className="faq-card">
            <button type="button" onClick={() => setActiveId(item.id)}>
              {item.question}
            </button>
            {activeId === item.id ? <p>{item.answer}</p> : null}
          </section>
        ))}
      </div>
    </main>
  );
}
```
