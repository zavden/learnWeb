## @file src/App.jsx
## @lang jsx
## @role app

```jsx
import React from 'react';
import glossary from './data/glossary.json';

export function App() {
  return (
    <main className="glossary-demo">
      <p className="eyebrow">Project JSON</p>
      <h1>{glossary.title}</h1>
      <div className="glossary-grid">
        {glossary.entries.map((entry) => (
          <article key={entry.term} className="entry-card">
            <h2>{entry.term}</h2>
            <p>{entry.definition}</p>
          </article>
        ))}
      </div>
    </main>
  );
}
```
