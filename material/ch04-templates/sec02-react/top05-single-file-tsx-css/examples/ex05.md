# TSX
```tsx
type Mood = 'focused' | 'resting' | 'shipping';

function App() {
  const [mood, setMood] = React.useState<Mood>('focused');

  React.useEffect(() => {
    console.info('mood', mood);
  }, [mood]);

  return (
    <main className="mood-card">
      <p className="eyebrow">TSX + CSS</p>
      <h1>{mood}</h1>
      <div className="mood-row">
        <button type="button" onClick={() => setMood('focused')}>Focused</button>
        <button type="button" onClick={() => setMood('resting')}>Resting</button>
        <button type="button" onClick={() => setMood('shipping')}>Shipping</button>
      </div>
    </main>
  );
}
```
