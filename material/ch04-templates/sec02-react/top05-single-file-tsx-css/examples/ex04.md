# TSX
```tsx
type Step = {
  id: number;
  label: string;
};

const steps: Step[] = [
  { id: 1, label: 'Setup' },
  { id: 2, label: 'Code' },
  { id: 3, label: 'Review' },
];

function App() {
  const [activeStep, setActiveStep] = React.useState<number>(1);

  return (
    <main className="steps-card">
      <h1>Step {activeStep}</h1>
      <div className="step-list">
        {steps.map((step) => (
          <button key={step.id} type="button" onClick={() => setActiveStep(step.id)}>
            {step.label}
          </button>
        ))}
      </div>
    </main>
  );
}
```
