# TSX
```tsx
type Metric = {
  id: number;
  label: string;
  value: number;
};

const metrics: Metric[] = [
  { id: 1, label: 'Props', value: 4 },
  { id: 2, label: 'State', value: 5 },
  { id: 3, label: 'Effects', value: 3 },
];

function App() {
  const [selectedId, setSelectedId] = React.useState<number>(metrics[0].id);
  const selected = metrics.find((metric) => metric.id === selectedId) || metrics[0];

  return (
    <main className="metrics-card">
      <h1>{selected.label}</h1>
      <p>{selected.value} lessons</p>
      <div className="metric-grid">
        {metrics.map((metric) => (
          <button key={metric.id} type="button" onClick={() => setSelectedId(metric.id)}>
            {metric.label}
          </button>
        ))}
      </div>
    </main>
  );
}
```
