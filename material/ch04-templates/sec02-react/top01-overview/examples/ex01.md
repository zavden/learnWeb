# JSX
```jsx
function App() {
  const [count, setCount] = React.useState(0);

  return (
    <main>
      <p>React overview sampler</p>
      <h1>Single-file JSX</h1>
      <button type="button" onClick={() => setCount(count + 1)}>
        Clicked {count} times
      </button>
    </main>
  );
}
```
