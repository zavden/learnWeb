## @file src/App.jsx
## @lang jsx
## @role app

```jsx
import React from 'react';
import dataset from './data/modules.json';

export function App() {
  return (
    <main className="json-demo">
      <p className="eyebrow">Overview sampler</p>
      <h1>{dataset.title}</h1>
      <ul>
        {dataset.modules.map((module) => (
          <li key={module.id}>
            <strong>{module.label}</strong>
            <span>{module.level}</span>
          </li>
        ))}
      </ul>
    </main>
  );
}
```
