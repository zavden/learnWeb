# TSX
```tsx
type LessonStage = 'setup' | 'practice' | 'review';

function App() {
  const [stage, setStage] = React.useState<LessonStage>('setup');

  const message = {
    setup: 'Prepare the component and props.',
    practice: 'Experiment with state and events.',
    review: 'Inspect what the example taught you.',
  }[stage];

  return (
    <main>
      <p>React + TSX overview</p>
      <h1>{stage}</h1>
      <p>{message}</p>
      <button type="button" onClick={() => setStage('setup')}>Setup</button>
      <button type="button" onClick={() => setStage('practice')}>Practice</button>
      <button type="button" onClick={() => setStage('review')}>Review</button>
    </main>
  );
}
```
