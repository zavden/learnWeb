# HTML

```html
<button id="status-toggle" type="button">Set online</button>
<p id="status-text">Offline</p>
```

# TypeScript

```typescript
let online: boolean = false;
const button = document.getElementById('status-toggle') as HTMLButtonElement | null;
const text = document.getElementById('status-text') as HTMLParagraphElement | null;

button?.addEventListener('click', () => {
  online = !online;
  if (button && text) {
    button.textContent = online ? 'Set offline' : 'Set online';
    text.textContent = online ? 'Online' : 'Offline';
  }
});
```
