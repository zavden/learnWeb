  <p>{{ current.prompt }}</p>

    <button v-for="answer in current.answers" :key="answer" type="button" @click="select(answer)">

    </button>
  </div>
  <p class="quiz__result">{{ result }}</p>
</section>

# JavaScript

```javascript
import { computed, ref } from 'vue';

export default {
  setup() {
    const title = ref('Quiz card');
    const current = ref({
      prompt: 'Which API creates derived reactive values?',
      answers: ['watch', 'computed', 'emit'],
      correct: 'computed',
    });
    const picked = ref('');
    const result = computed(() => {
      if (!picked.value) {
        return 'Pick an answer.';
      }
      return picked.value === current.value.correct ? 'Correct.' : 'Try again.';
    });
    const select = (answer) => {
      picked.value = answer;
    };

    return {
      current,
      result,
      select,
      title,
    };
  },
};
```
