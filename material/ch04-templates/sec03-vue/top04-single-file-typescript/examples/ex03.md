# TypeScript

```typescript
import { defineComponent, ref } from 'vue';

type Student = {
  name: string;
  completed: number;
};

export default defineComponent({
  setup() {
    const title = ref<string>('Typed object state');
    const student = ref<Student>({
      name: 'Ari',
      completed: 2,
    });
    const finishLesson = () => {
      student.value = {
        ...student.value,
        completed: student.value.completed + 1,
      };
    };

    return {
      finishLesson,
      student,
      title,
    };
  },
});
```
