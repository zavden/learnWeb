# HTML

```html
<section>
  <h1>{{ title }}</h1>
  <p>{{ summary }}</p>
  <ul>
    <li v-for="item in checklist" :key="item.id">
      <label>
        <input v-model="item.done" type="checkbox" />
        {{ item.label }}
      </label>
    </li>
  </ul>
</section>
```

# TypeScript

```typescript
import { computed, defineComponent, ref } from 'vue';

type ChecklistItem = {
  id: number;
  label: string;
  done: boolean;
};

export default defineComponent({
  setup() {
    const title = ref<string>('Typed checklist');
    const checklist = ref<ChecklistItem[]>([
      { id: 1, label: 'Read the template', done: true },
      { id: 2, label: 'Trace the refs', done: false },
      { id: 3, label: 'Press the button', done: false },
    ]);
    const summary = computed(() => {
      const completed = checklist.value.filter((item) => item.done).length;
      return `${completed} of ${checklist.value.length} steps complete.`;
    });

    return {
      checklist,
      summary,
      title,
    };
  },
});
```
