# HTML

```html
<section>
  <h1>{{ title }}</h1>
  <label>
    Focus
    <select v-model="selectedId">
      <option v-for="lesson in lessons" :key="lesson.id" :value="lesson.id">{{ lesson.label }}</option>
    </select>
  </label>
  <p>{{ current.description }}</p>
</section>
```

# TypeScript

```typescript
import { computed, defineComponent, ref } from 'vue';

type Lesson = {
  id: string;
  label: string;
  description: string;
};

export default defineComponent({
  setup() {
    const title = ref<string>('Typed lesson picker');
    const lessons = ref<Lesson[]>([
      { id: 'refs', label: 'Refs', description: 'Mutable reactive state.' },
      { id: 'computed', label: 'Computed', description: 'Reactive derived values.' },
      { id: 'watch', label: 'Watch', description: 'React to state changes.' },
    ]);
    const selectedId = ref<string>('refs');
    const current = computed<Lesson>(() => (
      lessons.value.find((lesson) => lesson.id === selectedId.value) ?? lessons.value[0]
    ));

    return {
      current,
      lessons,
      selectedId,
      title,
    };
  },
});
```
