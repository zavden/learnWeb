## @file src/App.vue
## @lang vue
## @role app

```vue
<template>
  <main class="tab-shell">
    <div class="tab-shell__actions">
      <button v-for="tab in tabs" :key="tab.id" type="button" @click="selectTab(tab.id)">
        {{ tab.label }}
      </button>
    </div>
    <article class="tab-shell__panel">
      <h1>{{ current.label }}</h1>
      <p>{{ current.body }}</p>
    </article>
  </main>
</template>

<script setup>
import { computed, ref } from 'vue';

const tabs = [
  { id: 'template', label: 'Template', body: 'SFC templates stay close to the script.' },
  { id: 'logic', label: 'Logic', body: 'script setup keeps small examples compact.' },
  { id: 'style', label: 'Style', body: 'Local styles can live in the same component.' },
];

const currentId = ref('template');
const current = computed(() => tabs.find((tab) => tab.id === currentId.value) ?? tabs[0]);
const selectTab = (id) => {
  currentId.value = id;
};
</script>
```
