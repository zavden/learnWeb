## @file src/composables/useCards.js
## @lang javascript
## @role hook

```javascript
import { computed, ref } from 'vue';

export function useCards() {
  const cards = [
    { front: 'ref()', back: 'Store reactive state.' },
    { front: 'computed()', back: 'Derive values from reactive state.' },
    { front: 'watch()', back: 'Observe state changes.' },
  ];
  const index = ref(0);
  const current = computed(() => cards[index.value]);
  const nextCard = () => {
    index.value = (index.value + 1) % cards.length;
  };

  return {
    current,
    nextCard,
  };
}
```
