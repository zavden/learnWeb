## @file src/App.js
## @lang javascript
## @role app

```javascript
import { computed, defineComponent, ref } from 'vue';
import render from './App.html';

export default defineComponent({
  setup() {
    const tabs = ref([
      { id: 'intro', label: 'Intro', body: 'Start with refs and template bindings.' },
      { id: 'layout', label: 'Layout', body: 'Split markup and shared styles.' },
      { id: 'review', label: 'Review', body: 'Use multiple files when structure matters.' },
    ]);
    const currentId = ref('intro');
    const current = computed(() => (
      tabs.value.find((tab) => tab.id === currentId.value) ?? tabs.value[0]
    ));
    const selectTab = (id) => {
      currentId.value = id;
    };

    return {
      current,
      selectTab,
      tabs,
    };
  },
  render,
});
```

## @file src/App.html
## @lang html
## @role markup

```html
<main class="tabs-shell">
  <div class="tabs-shell__actions">
    <button v-for="tab in tabs" :key="tab.id" type="button" @click="selectTab(tab.id)">
      {{ tab.label }}
    </button>
  </div>
  <section class="tabs-shell__panel">
    <h1>{{ current.label }}</h1>
    <p>{{ current.body }}</p>
  </section>
</main>
```
