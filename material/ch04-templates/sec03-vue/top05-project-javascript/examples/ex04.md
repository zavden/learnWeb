## @file src/composables/useFlashcards.js
## @lang javascript
## @role hook

```javascript
import { computed, ref } from 'vue';

export function useFlashcards() {
  const cards = [
    { front: 'ref()', back: 'Stores reactive mutable state.' },
    { front: 'computed()', back: 'Builds a derived reactive value.' },
    { front: 'watch()', back: 'Responds to changes with side effects.' },
  ];
  const index = ref(0);
  const current = computed(() => cards[index.value]);
  const nextCard = () => {
    index.value = (index.value + 1) % cards.length;
  };

  return {
    current,
    nextCard,
  };
}
```
