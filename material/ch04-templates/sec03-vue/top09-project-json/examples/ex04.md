## @file src/App.ts
## @lang typescript
## @role app

```typescript
import { computed, defineComponent, ref } from 'vue';
import plans from './data/plans.json';
import render from './App.html';

export default defineComponent({
  setup() {
    const selected = ref<string>('starter');
    const current = computed(() => (
      plans.items.find((plan) => plan.id === selected.value) ?? plans.items[0]
    ));
    const selectPlan = (id: string) => {
      selected.value = id;
    };

    return {
      current,
      plans,
      selectPlan,
    };
  },
  render,
});
```

## @lang html

    </button>
  </div>
