## @file src/App.vue
## @lang vue
## @role app

```vue
<template>
  <main class="dataset-shell">
    <h1>{{ data.title }}</h1>
    <p>Total examples: {{ totalExamples }}</p>
  </main>
</template>

<script setup lang="ts">
import { computed } from 'vue';
import data from './data/curriculum.json';

const totalExamples = computed<number>(() => (
  data.sections.reduce((sum, section) => sum + section.examples, 0)
));
</script>
```
