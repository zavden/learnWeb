## @file src/App.vue
## @lang vue
## @role app

```vue
<template>
  <main class="progress-shell">
    <h1>{{ title }}</h1>
    <p>{{ summary }}</p>
    <button type="button" @click="advance">Advance</button>
  </main>
</template>

<script setup lang="ts">
import { computed, ref } from 'vue';

const title = 'Computed values in SFC TS';
const completed = ref<number>(2);
const total = 5;
const summary = computed(() => `Completed ${completed.value} of ${total} lessons.`);
const advance = () => {
  if (completed.value < total) {
    completed.value += 1;
  }
};
</script>
```
