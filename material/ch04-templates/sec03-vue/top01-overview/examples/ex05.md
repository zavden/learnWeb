## @file src/composables/useCounter.ts
## @lang typescript
## @role hook

```typescript
import { ref, type Ref } from 'vue';

export type CounterApi = {
  count: Ref<number>;
  increment: () => void;
};

export function useCounter(initialValue: number): CounterApi {
  const count = ref<number>(initialValue);

  return {
    count,
    increment: () => {
      count.value += 1;
    },
  };
}
```
