# TypeScript

```typescript
import { defineComponent, ref } from 'vue';

export default defineComponent({
  setup() {
    const title = ref<string>('Typed single-file mode');
    const moduleName = ref<string>('Vue TypeScript');
    const completed = ref<number>(1);
    const completeStep = () => {
      completed.value += 1;
    };

    return {
      completeStep,
      completed,
      moduleName,
      title,
    };
  },
});
```
