# JavaScript

```javascript
import { computed, ref } from 'vue';

export default {
  setup() {
    const bright = ref(true);
    const title = ref('Styled single-file mode');
    const actionLabel = computed(() => (bright.value ? 'Switch to calm' : 'Switch to bright'));
    const message = computed(() => (
      bright.value ? 'Template, CSS and script live together.' : 'Still reactive, just with a softer message.'
    ));
    const toggleTheme = () => {
      bright.value = !bright.value;
    };

    return {
      actionLabel,
      message,
      title,
      toggleTheme,
    };
  },
};
```
