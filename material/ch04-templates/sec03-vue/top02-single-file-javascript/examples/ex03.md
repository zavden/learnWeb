# HTML

```html
<section>
  <h1>{{ title }}</h1>
  <p>{{ currentLesson.description }}</p>
  <label>
    Lesson
    <select v-model="selectedId">
      <option v-for="lesson in lessons" :key="lesson.id" :value="lesson.id">{{ lesson.label }}</option>
    </select>
  </label>
</section>
```

# JavaScript

```javascript
import { computed, ref } from 'vue';

export default {
  setup() {
    const title = ref('Computed selections');
    const lessons = ref([
      { id: 'refs', label: 'Refs', description: 'Store reactive values.' },
      { id: 'computed', label: 'Computed', description: 'Derive reactive values.' },
      { id: 'events', label: 'Events', description: 'Respond to user input.' },
    ]);
    const selectedId = ref('refs');
    const currentLesson = computed(() => (
      lessons.value.find((lesson) => lesson.id === selectedId.value) ?? lessons.value[0]
    ));

    return {
      currentLesson,
      lessons,
      selectedId,
      title,
    };
  },
};
```
