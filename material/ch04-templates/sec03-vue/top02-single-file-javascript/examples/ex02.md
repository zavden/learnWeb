# HTML

```html
<main>
  <h1>{{ title }}</h1>
  <button type="button" @click="toggle">{{ buttonLabel }}</button>
  <p v-if="open">{{ answer }}</p>
</main>
```

# JavaScript

```javascript
import { computed, ref } from 'vue';

export default {
  setup() {
    const title = ref('Conditional rendering');
    const open = ref(false);
    const answer = ref('v-if swaps template branches based on reactive state.');
    const buttonLabel = computed(() => (open.value ? 'Hide note' : 'Show note'));
    const toggle = () => {
      open.value = !open.value;
    };

    return {
      answer,
      buttonLabel,
      open,
      title,
      toggle,
    };
  },
};
```
