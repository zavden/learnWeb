# HTML

```html
<section>
  <h1>{{ title }}</h1>
  <p>{{ description }}</p>
  <button type="button" @click="previous" :disabled="index === 0">Previous</button>
  <button type="button" @click="next" :disabled="index === steps.length - 1">Next</button>
</section>
```

# JavaScript

```javascript
import { computed, ref } from 'vue';

export default {
  setup() {
    const title = ref('Step navigation');
    const steps = ['Start', 'Template', 'Script', 'Finish'];
    const index = ref(0);
    const description = computed(() => `Current step: ${steps[index.value]}`);
    const previous = () => {
      index.value -= 1;
    };
    const next = () => {
      index.value += 1;
    };

    return {
      description,
      index,
      next,
      previous,
      steps,
      title,
    };
  },
};
```
