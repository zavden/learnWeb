## @file src/App.ts
## @lang typescript
## @role app

```typescript
import { computed, defineComponent, ref } from 'vue';
import render from './App.html';

type Module = {
  label: string;
  duration: number;
};

export default defineComponent({
  setup() {
    const modules = ref<Module[]>([
      { label: 'Intro', duration: 10 },
      { label: 'Template', duration: 15 },
      { label: 'Typing', duration: 20 },
    ]);
    const totalMinutes = computed<number>(() => (
      modules.value.reduce((sum, module) => sum + module.duration, 0)
    ));

    return {
      modules,
      totalMinutes,
    };
  },
  render,
});
```

## @file src/App.html
## @lang html
## @role markup

```html

  <ul>
    <li v-for="module in modules" :key="module.label">{{ module.label }} - {{ module.duration }} min</li>
  </ul>
  <p>Total: {{ totalMinutes }} minutes</p>
</main>
```
