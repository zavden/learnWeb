## @file src/composables/useSprint.ts
## @lang typescript
## @role hook

```typescript
import { computed, ref } from 'vue';

type Sprint = {
  title: string;
  goal: string;
};

export function useSprint() {
  const sprints: Sprint[] = [
    { title: 'Sprint 1', goal: 'Learn refs.' },
    { title: 'Sprint 2', goal: 'Split files.' },
    { title: 'Sprint 3', goal: 'Type the props.' },
  ];
  const index = ref<number>(0);
  const current = computed<Sprint>(() => sprints[index.value]);
  const nextSprint = () => {
    index.value = (index.value + 1) % sprints.length;
  };

  return {
    current,
    nextSprint,
  };
}
```
