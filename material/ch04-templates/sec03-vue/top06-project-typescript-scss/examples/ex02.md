## @file src/App.ts
## @lang typescript
## @role app

```typescript
import { computed, defineComponent, ref } from 'vue';
import render from './App.html';

type Track = {
  id: string;
  label: string;
  level: string;
};

export default defineComponent({
  setup() {
    const tracks = ref<Track[]>([
      { id: 'vanilla', label: 'Vanilla', level: 'Beginner' },
      { id: 'react', label: 'React', level: 'Intermediate' },
      { id: 'vue', label: 'Vue', level: 'Intermediate' },
    ]);
    const selectedId = ref<string>('vanilla');
    const current = computed<Track>(() => (
      tracks.value.find((track) => track.id === selectedId.value) ?? tracks.value[0]
    ));
    const selectTrack = (id: string) => {
      selectedId.value = id;
    };

    return {
      current,
      selectTrack,
      tracks,
    };
  },
  render,
});
```

## @lang html
## @role markup

    </button>
  </div>
  <section class="track-picker__panel">
